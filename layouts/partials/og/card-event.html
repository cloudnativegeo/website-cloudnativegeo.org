{{- /* og/card-event.html — the inverted "event" Open Graph card (1200×630): the photo-negative of the Masthead card, and structurally consistent with it. Bonus Blue ground, Soft White top bar with a blue wordmark (left) + the events URL (right), a white auto-fit headline centered between the bar and the rule, and a white footer over a thin rule: the location (left) and the date with the time tucked under it (right) — the same left/right metadata grammar as the blog card. Static chrome (blue ground, white bar, blue wordmark, footer rule) is the committed assets/og/base-event.png. Flagship events may set front-matter `og_card_photo: ` to use a build-time Bonus-Blue duotone of that photo as the ground instead (see the background block below). Called with the page as context; returns the og.png resource. */ -}} {{- $page := . -}} {{- $m := site.Data.og_metrics -}} {{- /* ---- content ---- */ -}} {{- $title := $page.Title | plainify | htmlUnescape | chomp -}} {{- $url := "cloudnativegeo.org/events" -}} {{- $date := $page.Params.display_date | default "" -}} {{- $time := $page.Params.when_time | default "" -}} {{- /* location = the part of `where` before the street address; keep "Virtual" etc. */ -}} {{- $loc := "" -}} {{- with $page.Params.where -}} {{- $loc = index (split (replace . " – " " - ") " - ") 0 | chomp -}} {{- end -}} {{- /* ---- geometry (px on the 1200×630 canvas) ---- */ -}} {{- $padX := 88 -}} {{- $usableW := 1024 -}} {{- $rightEdge := 1112 -}} {{- $hBoxTop := 156 -}} {{- /* headline centered between the bar and the rule (156 → 500) */ -}} {{- $hBoxH := 344 -}} {{- $footY := 518 -}} {{- /* location + date baseline */ -}} {{- $footSize := 28 -}} {{- $timeY := 548 -}} {{- /* time, tucked under the date */ -}} {{- $timeSize := 20 -}} {{- $white := "#F2F4F6" -}} {{- $blue := "#2126F7" -}} {{- $dim := "#B7BBF6" -}} {{- /* ~72% white over Bonus Blue, for the time */ -}} {{- $quattro := resources.Get "/fonts/iAWriterQuattroS-Bold.ttf" -}} {{- $monoBold := partial "og/mono.html" "Bold" -}} {{- $monoReg := partial "og/mono.html" "Regular" -}} {{- /* Background: flat Bonus Blue by default, or a build-time Bonus-Blue duotone of a front-matter photo for flagship events: og_card_photo: images/250603-cng-conf-open.jpg og_card_treatment: duotone # optional; duotone is the only treatment The photo is cropped to 1200×630, desaturated, recolored blue, then the chrome+legibility-gradient overlay is composited on top. */ -}} {{- $base := resources.Get "og/base-event.png" -}} {{- with $page.Params.og_card_photo -}} {{- $treatment := $page.Params.og_card_treatment | default "duotone" -}} {{- with resources.Get . -}} {{- if eq $treatment "duotone" -}} {{- $ground := (.Fill "1200x630 Smart").Filter images.Grayscale (images.Colorize 239 78 64) -}} {{- $base = $ground.Filter (images.Overlay (resources.Get "og/base-event-chrome.png") 0 0) -}} {{- end -}} {{- else -}} {{- warnf "OG event card: og_card_photo %q not found for %s" . $page.RelPermalink -}} {{- end -}} {{- end -}} {{- /* "EVENT" eyebrow (Sunshine Yellow) + white headline, centered as a group */ -}} {{- $filters := partial "og/headline.html" (dict "title" $title "m" $m "font" $quattro "color" $white "eyebrow" "EVENT" "eyeColor" "#FFD626" "eyeFont" $monoBold "padX" $padX "usableW" $usableW "boxTop" $hBoxTop "boxH" $hBoxH "min" 34 "max" 92 "pitch" 1.04 "eyeSize" 24 "eyeGap" 16) -}} {{- /* URL — Berkeley Mono Regular 24, blue, right-aligned in the white bar */ -}} {{- with $monoReg -}} {{- $w := mul (mul (len $url) $m.mono.adv) 24 -}} {{- $filters = $filters | append (images.Text $url (dict "color" $blue "size" 24 "font" . "x" (int (sub $rightEdge $w)) "y" 44 )) -}} {{- end -}} {{- /* Footer (mirrors the blog card): location bottom-left; date bottom-right with the time tucked under it. Location truncates so it never hits the date. */ -}} {{- if $monoBold -}} {{- $dateW := mul (mul (len $date) $m.mono.adv) $footSize -}} {{- $timeW := mul (mul (len $time) $m.mono.adv) $timeSize -}} {{- with $date -}} {{- $filters = $filters | append (images.Text . (dict "color" $white "size" $footSize "font" $monoBold "x" (int (sub $rightEdge $dateW)) "y" $footY )) -}} {{- end -}} {{- with $time -}} {{- $filters = $filters | append (images.Text . (dict "color" $dim "size" $timeSize "font" $monoReg "x" (int (sub $rightEdge $timeW)) "y" $timeY )) -}} {{- end -}} {{- with $loc -}} {{- $charW := mul $m.mono.adv $footSize -}} {{- $rightW := math.Max $dateW $timeW -}} {{- $avail := sub (sub (sub $rightEdge $padX) $rightW) 28.0 -}} {{- $maxChars := int (math.Floor (div $avail $charW)) -}} {{- $txt := . -}} {{- if gt (len $txt) $maxChars -}}{{- $txt = printf "%s…" (substr $txt 0 (sub $maxChars 1)) -}}{{- end -}} {{- $filters = $filters | append (images.Text $txt (dict "color" $white "size" $footSize "font" $monoBold "x" $padX "y" $footY )) -}} {{- end -}} {{- end -}} {{- $img := $base.Filter $filters -}} {{- $img = resources.Copy (path.Join $page.RelPermalink "og.png") $img -}} {{- return $img -}}
